─── generate_data.py ────────────────────────────────────────────────────────

import numpy as np, pandas as pd
np.random.seed(42)
data = []

# 333 cheap_hunter samples
for _ in range(333):
    avg_price      = np.random.uniform(5, 199)
    purchase_count = np.random.randint(1, 20)
    total_spent    = avg_price * purchase_count * np.random.uniform(0.8, 1.2)
    fav_cat        = np.random.choice([0, 2], p=[0.2, 0.8])
    data.append([avg_price, total_spent, purchase_count, fav_cat, 0])

# 334 normal_user samples
for _ in range(334):
    avg_price      = np.random.uniform(200, 999)
    purchase_count = np.random.randint(1, 15)
    total_spent    = avg_price * purchase_count * np.random.uniform(0.8, 1.2)
    fav_cat        = np.random.choice([0, 1, 2], p=[0.5, 0.2, 0.3])
    data.append([avg_price, total_spent, purchase_count, fav_cat, 1])

# 333 premium_user samples
for _ in range(333):
    avg_price      = np.random.uniform(1000, 3000)
    purchase_count = np.random.randint(1, 10)
    total_spent    = avg_price * purchase_count * np.random.uniform(0.8, 1.2)
    fav_cat        = np.random.choice([0, 1], p=[0.5, 0.5])
    data.append([avg_price, total_spent, purchase_count, fav_cat, 2])

df = pd.DataFrame(data, columns=[
    'avg_price','total_spent','purchase_count','fav_category','label'])
df.sample(frac=1).reset_index(drop=True).to_csv('training_data.csv', index=False)

─── model.py (Keras Neural Network) ─────────────────────────────────────────

import tensorflow as tf
from tensorflow.keras import layers, models

def build_model():
    model = models.Sequential([
        layers.Input(shape=(4,)),
        layers.Dense(64, activation='relu'),
        layers.BatchNormalization(),
        layers.Dropout(0.3),
        layers.Dense(32, activation='relu'),
        layers.BatchNormalization(),
        layers.Dropout(0.2),
        layers.Dense(16, activation='relu'),
        layers.Dense(3, activation='softmax')   # 3 output classes
    ])
    model.compile(
        optimizer='adam',
        loss='sparse_categorical_crossentropy',
        metrics=['accuracy']
    )
    return model

─── train.py ─────────────────────────────────────────────────────────────────

import pandas as pd, numpy as np, joblib
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
from model import build_model

df = pd.read_csv('training_data.csv')
X  = df[['avg_price','total_spent','purchase_count','fav_category']].values
y  = df['label'].values

scaler  = StandardScaler()
X_scaled = scaler.fit_transform(X)
joblib.dump(scaler, 'scaler.pkl')

X_train, X_test, y_train, y_test = train_test_split(
    X_scaled, y, test_size=0.2, random_state=42)

model = build_model()
model.fit(X_train, y_train, epochs=50, batch_size=32,
          validation_split=0.2, verbose=1)

loss, acc = model.evaluate(X_test, y_test)
print(f"Test Accuracy: {acc:.4f}")   # Ket qua ky vong: ~95%

model.save('behavior_model.h5')

─── predict.py ───────────────────────────────────────────────────────────────

import numpy as np, tensorflow as tf, joblib

LABELS = ['cheap_hunter', 'normal_user', 'premium_user']
CAT_MAP = {'phone': 0, 'laptop': 1, 'accessory': 2}

model  = tf.keras.models.load_model('behavior_model.h5')
scaler = joblib.load('scaler.pkl')

def predict_segment(avg_price, total_spent, purchase_count, fav_category):
    cat_enc = CAT_MAP.get(fav_category, 2)
    X = np.array([[avg_price, total_spent, purchase_count, cat_enc]])
    probs = model.predict(scaler.transform(X), verbose=0)[0]
    idx = int(np.argmax(probs))
    return {
        'segment':    LABELS[idx],
        'confidence': float(probs[idx]),
        'scores': {
            'cheap_hunter': float(probs[0]),
            'normal_user':  float(probs[1]),
            'premium_user': float(probs[2])
        }
    }

─── build_index.py ───────────────────────────────────────────────────────────

from sentence_transformers import SentenceTransformer
import json, pickle

with open('kb_documents.json', 'r', encoding='utf-8') as f:
    documents = json.load(f)

embedder = SentenceTransformer('paraphrase-multilingual-MiniLM-L12-v2')
texts    = [f"{d['title']}. {d['content']}" for d in documents]
embeddings = embedder.encode(texts, show_progress_bar=True)

with open('kb_vectors.pkl', 'wb') as f:
    pickle.dump({'documents': documents, 'embeddings': embeddings}, f)

print(f"Built index for {len(documents)} documents")

━━━ 4.3 RAG PIPELINE ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

LUONG RAG:
  [1] User gui message
  [2] Embed cau hoi bang sentence-transformer
  [3] Cosine similarity voi kb_vectors.pkl
  [4] Lay top-3 documents lien quan nhat
  [5] Lay user segment tu DB (da classify boi DL model)
  [6] Tao prompt: [system] + [KB context] + [segment] + [cau hoi]
  [7] Goi Gemini API -> nhan response
  [8] Tra ve cau tra loi + sources cho frontend hien thi

─── retriever.py ─────────────────────────────────────────────────────────────

import pickle, numpy as np
from sentence_transformers import SentenceTransformer
from sklearn.metrics.pairwise import cosine_similarity

embedder = SentenceTransformer('paraphrase-multilingual-MiniLM-L12-v2')
with open('knowledge_base/kb_vectors.pkl', 'rb') as f:
    index = pickle.load(f)

def retrieve(query: str, segment: str = None, top_k: int = 3):
    query_vec = embedder.encode([query])
    sims = cosine_similarity(query_vec, index['embeddings'])[0]
    ranked = np.argsort(sims)[::-1]
    results = []
    for idx in ranked:
        doc   = index['documents'][idx]
        score = float(sims[idx])
        if segment and doc.get('segment') == segment:
            score += 0.1    # boost score neu dung segment
        results.append({'doc': doc, 'score': score})
    results.sort(key=lambda x: x['score'], reverse=True)
    return [r['doc'] for r in results[:top_k]]

─── generator.py ─────────────────────────────────────────────────────────────

import google.generativeai as genai, os

genai.configure(api_key=os.getenv('GEMINI_API_KEY'))
gemini = genai.GenerativeModel('gemini-1.5-flash')

SEG_DESC = {
    'cheap_hunter': 'khach hang ua thich san pham gia re, tiet kiem',
    'normal_user':  'khach hang thong thuong, can bang gia-chat luong',
    'premium_user': 'khach hang cao cap, ua thich san pham dinh nhat'
}

def generate(query: str, context_docs: list, segment: str) -> str:
    context = "\n\n".join([
        f"[{d['title']}]\n{d['content']}" for d in context_docs])
    prompt = f"""Ban la tro ly tu van mua sam dien tu cua NEXUS Store.

KHACH HANG: {segment} - {SEG_DESC.get(segment, '')}

TAI LIEU TU VAN (Knowledge Base):
{context}

CAU HOI: {query}

Tra loi ngan gon 3-5 cau, than thien, bang tieng Viet.
Goi y san pham phu hop voi phan loai khach hang."""
    try:
        return gemini.generate_content(prompt).text
    except:
        # fallback neu khong co API key
        titles = [d['title'] for d in context_docs[:2]]
        return f"Goi y danh cho ban: {', '.join(titles)}. Xem ngay tai NEXUS Store!"

─── pipeline.py ──────────────────────────────────────────────────────────────

from rag.retriever import retrieve
from rag.generator import generate

def rag_chat(query: str, segment: str) -> dict:
    relevant_docs = retrieve(query, segment=segment, top_k=3)
    answer        = generate(query, relevant_docs, segment)
    return {
        'answer':  answer,
        'segment': segment,
        'sources': [d['title'] for d in relevant_docs],
        'doc_ids': [d['id']    for d in relevant_docs]
    }
